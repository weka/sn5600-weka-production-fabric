ROCE_RUNTIME_AUDIT="roce_runtime_audit_64_77_$(date +%Y%m%d_%H%M%S).txt"

{
    for H in $(seq 64 77)
    do
        echo
        echo "================================================"
        echo "weka${H} 172.31.18.${H}"
        echo "================================================"

        ssh -o ConnectTimeout=7 -tt "root@172.31.18.${H}" '
            echo "hostname=$(hostname)"
            echo "tcp_ecn=$(sysctl -n net.ipv4.tcp_ecn 2>/dev/null || echo unknown)"

            echo "--- RDMA mapping and RoCE settings ---"

            ibdev2netdev 2>/dev/null || true

            while read -r IBDEV _ _ _ NETDEV _
            do
                [ -n "$IBDEV" ] || continue
                [ -n "$NETDEV" ] || continue

                echo
                echo "### $IBDEV / $NETDEV"

                ip -br link show "$NETDEV" 2>/dev/null || true

                ethtool "$NETDEV" 2>/dev/null |
                    grep -E "Speed:|Auto-negotiation:|Link detected:" ||
                    true

                printf "mtu="
                cat "/sys/class/net/$NETDEV/mtu" 2>/dev/null || echo unknown

                printf "roce_mode="
                cma_roce_mode -d "$IBDEV" -p 1 2>/dev/null || echo unknown

                printf "traffic_class="
                cat "/sys/class/infiniband/$IBDEV/tc/1/traffic_class" \
                    2>/dev/null || echo unknown

                mlnx_qos -i "$NETDEV" 2>/dev/null |
                    grep -E \
                    "DCBX mode:|Priority trust state:|PFC configuration:|^[[:space:]]*enabled" ||
                    true
            done < <(ibdev2netdev 2>/dev/null)
        ' || echo "UNREACHABLE: weka${H}"
    done
} 2>&1 | tee "$ROCE_RUNTIME_AUDIT"

echo "$HOME/Downloads/$ROCE_RUNTIME_AUDIT"
