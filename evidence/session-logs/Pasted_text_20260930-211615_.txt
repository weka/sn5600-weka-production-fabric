ROCE_AUDIT="roce_persistence_audit_64_77_$(date +%Y%m%d_%H%M%S).txt"

{
    for H in $(seq 64 77)
    do
        echo
        echo "================================================"
        echo "weka${H} 172.31.18.${H}"
        echo "================================================"

        ssh -o ConnectTimeout=7 -tt "root@172.31.18.${H}" '
            echo "hostname=$(hostname)"

            printf "tcp_ecn_runtime="
            sysctl -n net.ipv4.tcp_ecn 2>/dev/null || echo unknown

            printf "roce_service_enabled="
            systemctl is-enabled weka-roce.service 2>/dev/null || echo not-installed

            printf "roce_service_active="
            systemctl is-active weka-roce.service 2>/dev/null || echo inactive-or-missing

            printf "weka_agent_active="
            systemctl is-active weka-agent.service 2>/dev/null || echo inactive-or-missing

            echo "--- RoCE configuration ---"
            if [ -r /etc/weka-roce.conf ]; then
                cat /etc/weka-roce.conf
            else
                echo "NOT INSTALLED"
            fi

            echo "--- Persistent TCP ECN file ---"
            if [ -r /etc/sysctl.d/99-weka-roce.conf ]; then
                cat /etc/sysctl.d/99-weka-roce.conf
            else
                echo "NOT INSTALLED"
            fi

            echo "--- WEKA agent dependency ---"
            if [ -r /etc/systemd/system/weka-agent.service.d/10-roce.conf ]; then
                cat /etc/systemd/system/weka-agent.service.d/10-roce.conf
            else
                echo "NOT INSTALLED"
            fi

            echo "--- Generated configuration script TCP ECN lines ---"
            grep -n -C 1 "tcp_ecn" \
                /usr/local/sbin/weka-roce-config.sh 2>/dev/null ||
                echo "NO TCP ECN LINES OR SCRIPT NOT INSTALLED"

            echo "--- Service result ---"
            systemctl show weka-roce.service \
                -p ActiveState \
                -p SubState \
                -p Result \
                --no-pager 2>/dev/null || true
        ' || echo "UNREACHABLE: weka${H}"
    done
} 2>&1 | tee "$ROCE_AUDIT"

echo "$HOME/Downloads/$ROCE_AUDIT"
